OIDC Vue 集成示例

本文档展示如何在 Vue 2 和 Vue 3 项目中集成 OIDC 联合登录组件。

安装方式

<!-- 1. 引入 OIDC 基础库 -->
<script src="oidc.js"></script>
<link rel="stylesheet" href="oidc.css">

<!-- 2. 引入 Vue -->
<script src="https://unpkg.com/vue@3/dist/vue.global.js"></script>

<!-- 3. 引入 OIDC Vue 集成模块（必须在 Vue 之后引入）-->
<script src="oidc.vue.js"></script>

Vue 3 使用方式

方式一：使用插件（推荐）

const { createApp } = Vue;

const app = createApp({
    template: '<Oidc app-key="your-app-key" @success="onSuccess" />',
    methods: {
        onSuccess(user) {
            console.log('登录成功:', user);
        }
    }
});

// 安装 OIDC Vue 插件
app.use(oidcVue);

app.mount('#app');

方式二：使用组合式 API

const { createApp, ref, onMounted, onUnmounted } = Vue;

const app = createApp({
    setup() {
        // 使用 OIDC 组合式函数
        const { ospList, loading, init, login, dispose } = oidcVue.useVue();

        onMounted(() => {
            // 初始化 OIDC
            init('your-app-key', {
                mode: 'spa',
                target: 'dialog',
                success: (user) => {
                    console.log('登录成功:', user);
                }
            });
        });

        onUnmounted(() => {
            // 清理资源
            dispose();
        });

        return {
            ospList,
            loading,
            login
        };
    },
    template: `
        <div>
            <div v-if="loading">加载中...</div>
            <div v-else>
                <button v-for="osp in ospList"
                        :key="osp.code"
                        @click="login(osp.code)">
                    {{ osp.name }}
                </button>
            </div>
        </div>
    `
});

app.mount('#app');

方式三：全局属性访问

const app = createApp({
    mounted() {
        // 通过全局属性访问
        this.$oidc.init('your-app-key', 'container-id', {
            success: (user) => {
                console.log('登录成功:', user);
            }
        });
    }
});

app.use(oidcVue);
app.mount('#app');

Vue 2 使用方式

方式一：使用插件

// Vue 2 会自动检测并注册为 oidc 组件
Vue.use(oidcVue);

new Vue({
    el: '#app',
    template: '<oidc app-key="your-app-key" @success="onSuccess" />',
    methods: {
        onSuccess(user) {
            console.log('登录成功:', user);
        }
    }
});

方式二：手动注册组件

Vue.component('my-oidc', oidcVue.getVueComponent(Vue));

new Vue({
    el: '#app',
    template: '<my-oidc app-key="your-app-key" />'
});

方式三：通过原型访问

Vue.use(oidcVue);

new Vue({
    el: '#app',
    mounted() {
        this.$oidc.init('your-app-key', 'container-id', {
            mode: 'web',
            style: 'card',
            columns: 3
        });
    }
});

组件属性 (Props)

<Oidc
    app-key="your-app-key"        // 必填：应用Key
    mode="web"                     // 模式：web | spa
    view="list"                    // 视图：list | none
    style="item"                   // 样式：item | card | icon
    target="link"                  // 打开方式：link | tab | window | dialog
    https="false"                  // 是否使用HTTPS
    show-card="true"               // 是否显示卡片容器
    show-head="false"              // 是否显示头部
    head-html="<h3>登录</h3>"     // 头部HTML
    show-foot="false"              // 是否显示底部
    foot-html="<p>OIDC</p>"      // 底部HTML
    align="start"                  // 图标对齐：start | center | end
    columns="1"                    // 卡片列数
    response-type="code"           // 响应类型
    redirect-uri=""                // 回调地址
    state=""                       // 自定义状态
    scope=""                       // 授权范围
    @success="onSuccess"           // 登录成功事件
    @error="onError"               // 错误事件
/>

组合式 API 返回值

const {
    ospList,      // Ref<Array> - OSP列表
    loading,      // Ref<Boolean> - 加载状态
    error,        // Ref<String|null> - 错误信息
    initialized,  // Ref<Boolean> - 是否已初始化
    init,         // Function - 初始化方法
    login,        // Function - 登录方法（跳转）
    loginA,       // Function - SPA模式登录A
    loginB,       // Function - SPA模式登录B
    dispose,      // Function - 清理资源
    toUri,        // Function - 跳转到授权URL
    version       // String - 版本号
} = oidcVue.useVue();
