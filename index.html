OIDC 联合登录 - 使用示例

展示 OIDC SDK 的各种使用方式和配置选项

📖 基础用法

1. 引入 SDK

<!-- 引入CSS -->
<link rel="stylesheet" href="oidc.css" />

<!-- 引入SDK -->
<script src="oidc.js">

<!-- 容器 -->
<div id="oidc"></div>

<!-- 初始化 -->
<script>
    oidc.init('your-app-key', 'oidc', {
        mode: 'web',
        target: 'link'
    });
</script>

2. 初始化配置项说明

{
    // 必填项
    'appKey': 'your-app-key', // 应用密钥

    // 模式配置
    'mode': 'web', // web | spa (单页应用)
    'target': 'link', // link | tab | window | dialog

    // UI配置
    'view': 'list', // list | none (是否显示UI)
    'style': 'icon', // icon | card | item | list
    'columns': 5, // 卡片模式列数
    'align': 'center', // 图标对齐: start | center | end

    // 认证配置
    'response_type': 'code', // 响应类型
    'redirect_uri': 'https://...', // 回调地址
    'state': 'login', // 状态参数
    'scope': 'openid profile', // 授权范围

    // 日志配置
    'log': true, // 是否开启日志
    'logLevel': 'debug', // debug | info | warn | error

    // 安全配置
    'https': true // 是否使用HTTPS
}

⚙️ 动态配置

🚀 API 调用

1. 预加载数据

// 预加载服务列表
oidc.load('your-app-key', function(ospList) {
    console.log('服务列表:', ospList);
    // 后续可以多次调用 init
});

2. 直接登录

// Web模式登录
oidc.loginA('wechat'); // 微信登录
oidc.loginA('phone'); // 手机登录
oidc.loginA('email'); // 邮箱登录

// SPA模式登录
oidc.loginB('wechat');

3. 授权流程

// Web模式授权
oidc.authorizeA('wechat');

// SPA模式授权
oidc.authorizeB();

4. 状态控制

// 检查任务是否运行中
oidc.isRunning();

// 停止监听
oidc.endListen();
